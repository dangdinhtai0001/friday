// src/types/form.ts

// Định nghĩa kiểu dữ liệu cho dữ liệu đầu vào của form
export interface ProductInput {
  productName: string;
  productCode: string;
  price: number | null;
  category: string;
  isAvailable: boolean;
  deliveryOption: 'pickup' | 'delivery';
  deliveryAddress: string;
  notes: string;
}

// Định nghĩa kiểu cho lỗi validation
export interface FieldError {
  message: string;
}

export interface FormErrors<T> {
  [K in keyof T]?: FieldError;
}

// Định nghĩa kiểu cho phản hồi của hàm validate
export interface ValidateResponse<T> {
  values: Partial<T>; // Dữ liệu hợp lệ (có thể là một phần nếu có lỗi)
  errors: FormErrors<T>; // Các lỗi
}

// Định nghĩa các quy tắc validation cho từng trường
export interface FieldValidationRules {
  required?: boolean;
  minLength?: number;
  maxLength?: number;
  min?: number;
  max?: number;
  pattern?: RegExp;
  // Thêm các quy tắc validation khác nếu cần
}

// Định nghĩa kiểu cho cấu hình của một trường (field)
export interface FormFieldConfig<T extends ProductInput> {
  label: string;
  type: 'text' | 'number' | 'select' | 'checkbox' | 'radio' | 'textarea';
  placeholder?: string;
  options?: { value: string; label: string }[];
  validation?: FieldValidationRules;
  // Hàm callback khi giá trị của trường thay đổi
  onValueChange?: (newValue: T[keyof T], currentFormState: T) => void;
  // Các hàm để xác định trạng thái của trường
  isVisible?: (currentFormState: T) => boolean;
  isDisabled?: (currentFormState: T) => boolean;
  isReadOnly?: (currentFormState: T) => boolean;
}

// Định nghĩa kiểu cho toàn bộ cấu hình của form
export interface ProductFormConfigType {
  initialValues: ProductInput;
  fields: {
    [K in keyof ProductInput]: FormFieldConfig<ProductInput>;
  };
  // Hàm async để khởi tạo dữ liệu form
  asyncInitData: () => Promise<ProductInput>;
  // Hàm xử lý sự kiện khi form đã sẵn sàng
  onFormReady?: (currentFormState: ProductInput) => void;
  // Hàm validate toàn bộ form
  validateForm: (data: ProductInput) => Promise<ValidateResponse<ProductInput>>;
  // Hàm xử lý khi form được submit
  onSubmit: (formData: ProductInput, formState: ProductInput) => void; // formState có thể bao gồm cả metadata nếu bạn muốn
}

      ],
      validation: {

      },
      onValueChange: (value, formState) => {
        console.log(`Danh mục thay đổi thành: ${value}`);
      },
    },
    isAvailable: {
      label: 'Sản phẩm có sẵn',
      type: 'checkbox',
      onValueChange: (value, formState) => {

      },
    },
    deliveryOption: {
      label: 'Tùy chọn giao hàng',
      type: 'radio',
      options: [
        { value: 'pickup', label: 'Nhận tại cửa hàng' },
        { value: 'delivery', label: 'Giao hàng tận nơi' },

          productName: 'Điện Thoại Thông Minh XYZ',

          category: 'electronics',

          deliveryOption: 'delivery',

          notes: 'Sản phẩm mẫu được load từ hàm asyncInitData.',
        });
      }, 1500); // Giả lập độ trễ 1.5 giây
    });
  },

  // Hàm xử lý sự kiện khi form đã sẵn sàng
  onFormReady: (currentFormState) => {
    console.log('🚀 Form sản phẩm đã sẵn sàng! Dữ liệu hiện tại:', currentFormState);
    // Ví dụ: có thể set focus vào trường productName
    // document.getElementById('productName')?.focus();
  },

  // Hàm validate toàn bộ form (theo yêu cầu của bạn)
  validateForm: async (data: ProductInput): Promise<ValidateResponse<ProductInput>> => {
    console.log('Bắt đầu validate form với dữ liệu:', data);
    const errors: FormErrors<ProductInput> = {};

    // Validate theo các quy tắc trong schema
    for (const fieldName in productFormConfig.fields) {
      const fieldConfig = productFormConfig.fields[fieldName as keyof ProductInput];
      const value = data[fieldName as keyof ProductInput];
      const rules = fieldConfig.validation;

      if (!rules) continue;

      if (rules.required && (value === null || value === undefined || (typeof value === 'string' && value.trim() === ''))) {
        errors[fieldName as keyof ProductInput] = { message: `${fieldConfig.label} là bắt buộc.` };
      } else if (rules.minLength && typeof value === 'string' && value.length < rules.minLength) {
        errors[fieldName as keyof ProductInput] = { message: `${fieldConfig.label} phải có ít nhất ${rules.minLength} ký tự.` };
      } else if (rules.maxLength && typeof value === 'string' && value.length > rules.maxLength) {
        errors[fieldName as keyof ProductInput] = { message: `${fieldConfig.label} không được vượt quá ${rules.maxLength} ký tự.` };
      } else if (rules.min !== undefined && typeof value === 'number' && value < rules.min) {
        errors[fieldName as keyof ProductInput] = { message: `${fieldConfig.label} phải lớn hơn hoặc bằng ${rules.min}.` };
      } else if (rules.max !== undefined && typeof value === 'number' && value > rules.max) {
        errors[fieldName as keyof ProductInput] = { message: `${fieldConfig.label} phải nhỏ hơn hoặc bằng ${rules.max}.` };
      } else if (rules.pattern && typeof value === 'string' && !rules.pattern.test(value)) {
        errors[fieldName as keyof ProductInput] = { message: `${fieldConfig.label} không đúng định dạng.` };
      }
    }

    // Các validate logic phức tạp hơn, dựa trên giá trị của nhiều trường
    if (data.deliveryOption === 'delivery' && (data.deliveryAddress === null || data.deliveryAddress.trim() === '')) {
      errors.deliveryAddress = { message: 'Địa chỉ giao hàng là bắt buộc khi chọn giao hàng tận nơi.' };
    }

    // Ví dụ về validate tùy chỉnh như bạn yêu cầu:
    if (data.productName === '123') {
      errors.productName = { message: 'Tên sản phẩm "123" không được phép.' };
    }
    if (data.price !== null && data.price < 100000 && data.category === 'electronics') {
        errors.price = { message: 'Giá sản phẩm điện tử không thể thấp hơn 100,000 VND.' };
    }


    return {
      values: Object.keys(errors).length > 0 ? {} : data, // Nếu có lỗi, trả về values rỗng hoặc chỉ các giá trị hợp lệ
      errors,
    };
  },

  // Hàm xử lý khi form được submit
  onSubmit: (formData, formState) => {
    console.log('✅ Form đã được gửi!', formData);
    console.log('Trạng thái form đầy đủ khi gửi:', formState);
    // Đây là nơi bạn sẽ gọi API để lưu dữ liệu sản phẩm
    alert('Đã gửi dữ liệu sản phẩm! Kiểm tra console log.');
  },
